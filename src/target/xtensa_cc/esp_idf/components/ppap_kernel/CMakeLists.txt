# ppap_kernel component — PPAP kernel sources for ESP-IDF build
#
# Registers the PPAP kernel as an ESP-IDF component so that idf.py
# can build it alongside the ESP-IDF framework.

get_filename_component(PPAP_ROOT ${CMAKE_CURRENT_LIST_DIR}/../../../../../.. ABSOLUTE)

# Subsystems compiled into the kernel.  test_cpm / test_sos need both
# loaders plus the Z80 eCPU.  Human68k / m68k eCPU stay off pending
# separate validation.
set(PPAP_ENABLE_CPM       ON)
set(PPAP_ENABLE_SOS       ON)
set(PPAP_ENABLE_ECPU_Z80  ON)

include(${PPAP_ROOT}/cmake/kernel.cmake)

# ── Xtensa-CC target sources ─────────────────────────────────────────────────
set(XTENSA_CC_SOURCES
    ${PPAP_ROOT}/src/arch/xtensa/kernel/vfs/driver/usj.c
    ${PPAP_ROOT}/src/kernel/vfs/driver/fbcon.c
    ${PPAP_ROOT}/src/kernel/vfs/driver/lcd_st7789.c
    ${PPAP_ROOT}/src/target/xtensa_cc/kernel/vfs/driver/cc_kbd.c
    ${PPAP_ROOT}/src/target/xtensa_cc/kernel/vfs/driver/spi_lcd_xtensa_cc.c
    ${PPAP_ROOT}/src/target/xtensa_cc/kernel/vfs/driver/xtensa_cc_logger.c
    ${PPAP_ROOT}/src/target/xtensa_cc/kernel/core/xtensa_cc_stubs.c
)

# `driver` (legacy umbrella) provides spi_master + gpio for CC-4b's
# first-cut LCD transport.  CC-3.5e tracks the eventual lowering to
# bare MMIO, after which this dependency goes away.
idf_component_register(
    SRCS ${KERNEL_XTENSA_COMMON_SOURCES} ${XTENSA_CC_SOURCES}
    INCLUDE_DIRS
        "${PPAP_ROOT}/src"
        "${PPAP_ROOT}/src/arch/xtensa"
        "${PPAP_ROOT}/src/target/xtensa_cc"
    REQUIRES esp_psram driver
)

target_compile_definitions(${COMPONENT_LIB} PRIVATE
    PPAP_KERNEL=1
    PPAP_HAS_PROCFS=1
    PPAP_SYS_HZ=240000000u
    PROC_KSTACK_SIZE=2048u
    PAGE_POOL_BASE=0x3FC90000u
    PAGE_COUNT_MAX=48u
    MEM_REGION_RAM_TEXT_ARENA_SIZE=131072u
    MEM_REGION_EXT_TEXT_ARENA_SIZE=524288u
    MEM_REGION_EXT_RODATA_ARENA_SIZE=262144u
    ENABLE_XTENSA_PSRAM_EXEC=1
    PPAP_ENABLE_CPM=1
    PPAP_ENABLE_SOS=1
    PPAP_ENABLE_ECPU_Z80=1
)

# scripts/build.sh forwards --test as -DPPAP_TESTS=ON via idf.py.
# target_init_path() in target_xtensa_cc.c switches to /bin/runtests
# (or /bin/runtests_ext) when this is defined.
if(PPAP_TESTS)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE PPAP_TESTS=1)
endif()
if(PPAP_TESTS_EXTENDED)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE PPAP_TESTS_EXTENDED=1)
endif()

# ── Font generation pipeline (fbcon glyphs) ─────────────────────────────────
# fbcon.c references three font symbols (font8x16, font8x8, font4x8).  Even
# though xtensa_cc only uses MODE_SQUARE (8×8 → 30×16 on 240×135), the
# symbols are referenced unconditionally, so all three .c files must
# generate and link.  Mirrors pico1calc's pipeline; tools and inputs are
# shared.
find_package(Python3 REQUIRED COMPONENTS Interpreter)
file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/generated)

add_custom_command(
    OUTPUT ${CMAKE_BINARY_DIR}/generated/font8x16.c
    COMMAND ${Python3_EXECUTABLE} ${PPAP_ROOT}/tools/bdf2c.py
            --input ${PPAP_ROOT}/third_party/fonts/8x16.bdf
            --output ${CMAKE_BINARY_DIR}/generated/font8x16.c
            --name font8x16 --width 8 --height 16
    DEPENDS ${PPAP_ROOT}/third_party/fonts/8x16.bdf
            ${PPAP_ROOT}/tools/bdf2c.py
    COMMENT "Generating font8x16.c from 8x16.bdf"
)
add_custom_command(
    OUTPUT ${CMAKE_BINARY_DIR}/generated/font8x8.c
    COMMAND ${Python3_EXECUTABLE} ${PPAP_ROOT}/tools/json2c.py
            --input ${PPAP_ROOT}/third_party/fonts/ichigojam-font.json
            --output ${CMAKE_BINARY_DIR}/generated/font8x8.c
            --name font8x8
    DEPENDS ${PPAP_ROOT}/third_party/fonts/ichigojam-font.json
            ${PPAP_ROOT}/tools/json2c.py
    COMMENT "Generating font8x8.c from IchigoJam font"
)
add_custom_command(
    OUTPUT ${CMAKE_BINARY_DIR}/generated/font4x8.c
    COMMAND ${Python3_EXECUTABLE} ${PPAP_ROOT}/tools/bdf2c.py
            --input ${PPAP_ROOT}/third_party/fonts/4x6.bdf
            --output ${CMAKE_BINARY_DIR}/generated/font4x8.c
            --name font4x8 --width 4 --height 8
    DEPENDS ${PPAP_ROOT}/third_party/fonts/4x6.bdf
            ${PPAP_ROOT}/tools/bdf2c.py
    COMMENT "Generating font4x8.c from 4x6.bdf"
)

add_custom_target(generate_fonts
    DEPENDS ${CMAKE_BINARY_DIR}/generated/font8x16.c
            ${CMAKE_BINARY_DIR}/generated/font8x8.c
            ${CMAKE_BINARY_DIR}/generated/font4x8.c
)
add_dependencies(${COMPONENT_LIB} generate_fonts)

set_source_files_properties(
    ${CMAKE_BINARY_DIR}/generated/font8x16.c
    ${CMAKE_BINARY_DIR}/generated/font8x8.c
    ${CMAKE_BINARY_DIR}/generated/font4x8.c
    PROPERTIES GENERATED TRUE
)
target_sources(${COMPONENT_LIB} PRIVATE
    ${CMAKE_BINARY_DIR}/generated/font8x16.c
    ${CMAKE_BINARY_DIR}/generated/font8x8.c
    ${CMAKE_BINARY_DIR}/generated/font4x8.c
)

# ── Embed romfs.bin via .incbin ───────────────────────────────────────────────
# romfs.bin is generated by build.sh before idf.py runs.
# Generate an assembly wrapper that .incbin's it into .rodata.
set(ROMFS_BIN "${CMAKE_BINARY_DIR}/romfs.bin")
set(ROMFS_ASM "${CMAKE_BINARY_DIR}/romfs_data.S")

# Create a default empty romfs.bin if it doesn't exist yet (first configure)
if(NOT EXISTS "${ROMFS_BIN}")
    file(WRITE "${ROMFS_BIN}" "")
endif()

file(WRITE "${ROMFS_ASM}"
    "    .section .rodata.romfs, \"a\"\n"
    "    .global __romfs_start\n"
    "    .global __romfs_end\n"
    "__romfs_start:\n"
    "    .incbin \"${ROMFS_BIN}\"\n"
    "__romfs_end:\n"
)
target_sources(${COMPONENT_LIB} PRIVATE "${ROMFS_ASM}")

# Ensure ninja rebuilds romfs_data.S.obj when romfs.bin changes
# (.incbin dependencies are not tracked automatically)
set_source_files_properties("${ROMFS_ASM}" PROPERTIES
    OBJECT_DEPENDS "${ROMFS_BIN}")

# Warnings (do not add -Werror: ESP-IDF headers trigger warnings we can't fix)
target_compile_options(${COMPONENT_LIB} PRIVATE -Wno-unused-parameter)

# Override ESP-IDF's abort() and __assert_func() with PPAP handlers
# so kernel panics go through our fault handler with clear messages.
target_link_libraries(${COMPONENT_LIB} INTERFACE
    "-Wl,--wrap=abort"
    "-Wl,--wrap=__assert_func"
)
